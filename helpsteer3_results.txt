def is_valid_email(email):
    # Split the email into prefix and domain
    prefix, _, domain = email.partition('@')

    # Check if prefix is valid
    prefix_is_valid = True
    for char in prefix:
        if not (char.isalnum() or char in ['_', '.', '-']):
            prefix_is_valid = False
            break

    # Check if domain is valid
    domain_parts = domain.split('.')
    for part in domain_parts:
        is_valid = True
        for char in part:
            if not (char.isalnum() or char in ['_', '-']):
                is_valid = False
                break
        if not is_valid:
            break

        if len(part) > 1 and not (part[0] in ['_', '.', '-']):
            is_valid = False
            break

    # Check if domain has at least two parts
    if len(domain_parts) < 2:
        return False

    # Return True if both prefix and domain are valid, False otherwise
    return prefix_is_valid and is_valid

# Get email input from the user

**JIRA Plugin Development using Python**

In this example, we'll create a simple JIRA plugin using Python. This plugin will add a custom field to JIRA issues.

**Prerequisites:**

*   JIRA 8.13 or later
*   Python 3.8 or later
*   `jira` library (install using `pip install python-jira`)
*   A JIRA instance with an admin account

**Plugin Structure:**

Our plugin will consist of the following files:

*   `plugin.py`: The main plugin file
*   `plugin.xml`: The plugin descriptor file

**plugin.py**
```python
import json
import logging
from jira import JIRA
from jira.resources import Field

# Set up logging
logging.basicConfig(level=logging.INFO)

# JIRA credentials
jira_server = 'https://your-jira-instance.atlassian.net'
jira_username = 'your-jira-username'
jira_password = 'your-jira-password'

# Create a JIRA client
jira = JIRA(server=jira_server, basic_auth=(jira_username, jira_password))

# Define a custom field
class CustomField(Field):
    def __init__(self, name, description):
        self.name = name
        self.description = description

    def get_field(self, issue):
        return issue.fields.get_custom_field(self.name)

    def set_field(self, issue, value):
        issue.fields.customField = value

# Define a custom field instance
custom_field = CustomField('custom_field_name', 'Custom field description')

# Add a custom field to an issue
def add_custom_field(issue_id, value):
    issue = jira.issue(issue_id)
    custom_field.set_field(issue, value)
    jira.update_issue(issue, fields={'customField': custom_field.get_field(issue)})

# Example usage
issue_id = 'YOUR-12345'
value = 'Example value'
add_custom_field(issue_id, value)
```
